cmake_minimum_required(VERSION 3.24)
project(FakeClicker VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Concurrent Svg)
qt_standard_project_setup(REQUIRES 6.8)
if(MSVC)
 add_compile_options(/utf-8)
endif()
qt_add_executable(FakeClicker WIN32 src/main.cpp src/StartupController.cpp src/StartupController.h src/ClickController.cpp src/ClickController.h src/ClickWorker.cpp src/KeyBindings.cpp src/Configuration.cpp src/ConfigStore.cpp src/ConfigStore.h src/CaptureHarness.cpp src/CaptureHarness.h src/CrispIconProvider.h src/WindowPicker.cpp src/WindowPicker.h src/ClickPlanner.h src/WindowEffects.cpp src/WindowEffects.h)
set_target_properties(FakeClicker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
 enable_language(RC)
 add_executable(FakeClickerReceiver WIN32 tools/ClickReceiver.cpp)
 target_link_libraries(FakeClickerReceiver PRIVATE user32 gdi32)
 install(TARGETS FakeClickerReceiver RUNTIME DESTINATION .)
 if(MINGW)
  # Qt's MinGW entry point is built with stack protection.
  target_link_options(FakeClicker PRIVATE "-fstack-protector-strong")
  target_link_options(FakeClickerReceiver PRIVATE -municode)
 endif()
 target_sources(FakeClicker PRIVATE assets/windows.rc)
 set_property(SOURCE assets/windows.rc PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/assets")
 target_link_libraries(FakeClicker PRIVATE user32 shell32 dwmapi)
endif()
set_source_files_properties(qml/Theme.qml qml/I18n.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
file(GLOB APP_QML CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" qml/components/*.qml qml/components/*.js)
file(GLOB_RECURSE APP_ASSETS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" assets/*.svg assets/*.png assets/*.ttf assets/*.ico)
qt_add_qml_module(FakeClicker URI FakeClicker VERSION 1.0
 QML_FILES qml/Main.qml qml/Theme.qml qml/I18n.qml ${APP_QML}
 RESOURCES ${APP_ASSETS} assets/icons.json)
target_link_libraries(FakeClicker PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg)
if(MSVC)
 target_compile_options(FakeClicker PRIVATE /W4 /utf-8 /permissive-)
else()
 target_compile_options(FakeClicker PRIVATE -Wall -Wextra)
endif()
install(TARGETS FakeClicker RUNTIME DESTINATION .)
if(NOT CMAKE_CROSSCOMPILING)
 qt_generate_deploy_qml_app_script(TARGET FakeClicker OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
 install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
 qt_add_executable(controller_tests tests/controller_tests.cpp src/ClickController.cpp src/ClickController.h src/ClickWorker.cpp src/KeyBindings.cpp src/Configuration.cpp src/ConfigStore.cpp src/ConfigStore.h src/ClickPlanner.h)
 target_include_directories(controller_tests PRIVATE src)
 target_compile_definitions(controller_tests PRIVATE CCLICKER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(controller_tests PRIVATE Qt6::Gui Qt6::Concurrent)
 qt_add_executable(renderer_tests tests/renderer_tests.cpp)
 target_include_directories(renderer_tests PRIVATE src)
 target_compile_definitions(renderer_tests PRIVATE CCLICKER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(renderer_tests PRIVATE Qt6::Quick Qt6::Svg)
 add_test(NAME renderer_validation COMMAND renderer_tests)
 set_tests_properties(renderer_validation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 add_test(NAME controller_validation COMMAND controller_tests)
 set_tests_properties(controller_validation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_FOUND)
  add_test(NAME source_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_render_sources.py)
  add_test(NAME motion_contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_hover_regression.py)
 endif()
endif()
